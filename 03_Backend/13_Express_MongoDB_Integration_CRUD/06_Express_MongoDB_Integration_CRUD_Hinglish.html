💻 Complete Route Implementations
index.js — REST Routes
// 1. INDEX: Show all chats app.get("/chats", async (req, res) => { let chats = await Chat.find(); res.render("index.ejs", { chats }); }); // 2. UPDATE: PUT /chats/:id app.put("/chats/:id", async (req, res) => { let { id } = req.params; let { msg: newMsg } = req.body; await Chat.findByIdAndUpdate(id, { msg: newMsg }, { runValidators: true }); res.redirect("/chats"); }); // 3. DESTROY: DELETE /chats/:id app.delete("/chats/:id", async (req, res) => { let { id } = req.params; await Chat.findByIdAndDelete(id); res.redirect("/chats"); });
